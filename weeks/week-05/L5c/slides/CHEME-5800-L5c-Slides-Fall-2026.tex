% Notebook: Primal Linear Programming Problems.
\begin{frame}{Primal Linear Programming Problems}
  The \hilite{primal} describes the activities we choose: purchases,
  production levels, or edge flows.

  \vspace{0.7em}
  For $n$ activities and $m$ resource constraints, define:
  \begin{itemize}\setlength{\itemsep}{0.65em}
    \item $\mathbf{x}\in\mathbb R^n$: activity levels; $c_i$ is the objective
      contribution per unit of activity $i$.
    \item $\mathbf A\in\mathbb R^{m\times n}$: $A_{j,i}$ is the amount of
      resource $j$ used per unit of activity $i$.
    \item $\mathbf b\in\mathbb R^m$: $b_j$ is the available amount of resource $j$.
  \end{itemize}

  \vspace{0.6em}
  Matrix rows represent \hilite{resources}; columns represent \hilite{activities}.
  \lecturesource{Primal Linear Programming Problems: variables and indexing.}
\end{frame}

\begin{frame}{Maximize Value Within Resource Limits}
  With nonnegative activities, the resource-allocation model is:
  \[
    \begin{aligned}
      \underset{\mathbf x}{\text{maximize}}\quad
        & O(\mathbf x)=\mathbf c^\top\mathbf x
          =\sum_{i=1}^n c_i x_i\\
      \text{subject to}\quad & \mathbf A\mathbf x\leq\mathbf b,\\
        & \mathbf x\geq\mathbf0.
    \end{aligned}
  \]

  Each row limits total use of one resource:
  \[
    \underbrace{\sum_{i=1}^n A_{j,i}x_i}_{\text{resource }j\text{ used}}
    \leq\underbrace{b_j}_{\text{available}},\qquad j=1,\ldots,m.
  \]
  Coefficients are fixed; the contributions from activities add together.
  \lecturesource{Primal resource-allocation model and scalar resource constraint.}
\end{frame}

\begin{frame}{Feasible Does Not Mean Optimal}
  The \hilite{feasible region} contains all decisions satisfying every
  constraint and bound.

  \vspace{0.7em}
  \begin{itemize}\setlength{\itemsep}{0.65em}
    \item An \hilite{optimal} decision is feasible and attains the best
      objective value. Several decisions can share that value.
    \item An \hilite{infeasible} model has no decision satisfying all constraints.
    \item An \hilite{unbounded} maximization model has feasible decisions with
      arbitrarily large objective values, so no finite optimum exists.
  \end{itemize}

  \vspace{0.6em}
  Linear programs can also minimize an objective, use equality constraints,
  and impose other lower or upper bounds.
  \lecturesource{Feasibility, optimality, infeasibility, unboundedness, and other LP forms.}
\end{frame}

% Notebook: Consumer Choice Problems as Linear Programs.
\begin{frame}{Consumer Choice as a Linear Program}
  For product $i=1,\ldots,n$, let $x_i\geq0$ be the quantity purchased,
  $p_i>0$ its price in dollars per unit, and $u_i\geq0$ its utility per unit.
  Let $I\geq0$ be the budget in dollars.

  \vspace{0.4em}
  The consumer maximizes total utility subject to the budget:
  \[
    \begin{aligned}
      \underset{\mathbf x}{\text{maximize}}\quad
        & U(\mathbf x)=\sum_{i=1}^n u_i x_i\\
      \text{subject to}\quad & \sum_{i=1}^n p_i x_i\leq I,\\
        & x_i\geq0,\qquad i=1,\ldots,n.
    \end{aligned}
  \]
  One resource, the budget: $\mathbf c=\mathbf u$,
  $\mathbf A=[p_1\ \cdots\ p_n]$, and $\mathbf b=[I]$.

  \slidenote{Assumptions:}{divisible quantities, constant prices and utility per unit, and sufficient product availability.}
  \lecturesource{Consumer Choice Problems as Linear Programs.}
\end{frame}

% Notebook: Worked Allocation: Apples and Oranges.
\begin{frame}{Worked Allocation: Apples and Oranges}
  Let $x_A$ and $x_O$ be the quantities of apples and oranges.
  Each unit of apples costs 2 dollars and provides 0.55 utility units;
  each unit of oranges costs 4 dollars and provides 0.45 utility units.

  With a 100-dollar budget, the linear program is:
  \[
    \begin{aligned}
      \underset{x_A,x_O}{\text{maximize}}\quad
        & U(x_A,x_O)=0.55x_A+0.45x_O\\
      \text{subject to}\quad & 2x_A+4x_O\leq100,\\
        & x_A,x_O\geq0.
    \end{aligned}
  \]
  We can afford at most \hilite{50 apples} or \hilite{25 oranges}.
  Affordable mixtures are also feasible.
  \lecturesource{Worked Allocation: Apples and Oranges; first utility case.}
\end{frame}

\begin{frame}{Compare Purchases Along the Budget Boundary}
  Both fruits provide positive utility. With divisible quantities and no
  stock limit, spending unused budget improves utility. Thus, an optimum uses
  the full budget:
  \[
    x_O=25-\tfrac12x_A,\qquad 0\leq x_A\leq50.
  \]
  Substituting into the utility function gives:
  \[
    \begin{aligned}
      U(x_A)&=0.55x_A+0.45\left(25-\tfrac12x_A\right)\\
            &=11.25+\hilite{0.325}x_A.
    \end{aligned}
  \]
  Utility increases with $x_A$. The optimum is $(x_A^\star,x_O^\star)=(50,0)$,
  with $U^\star=27.5$. Buying only oranges gives 11.25 utility units.
  \lecturesource{Worked allocation: full-budget substitution and optimal purchase.}
\end{frame}

\begin{frame}{Compare Utility per Dollar}
  The ratio $u_i/p_i$ measures utility gained per dollar spent on product $i$.
  For the first case:
  \[
    \frac{u_A}{p_A}=\frac{0.55}{2}=\hilite{0.275},
    \qquad
    \frac{u_O}{p_O}=\frac{0.45}{4}=0.1125.
  \]

  \vspace{0.5em}
  \begin{itemize}\setlength{\itemsep}{0.8em}
    \item Moving one dollar from oranges to apples increases utility.
    \item Spending the whole budget does not, by itself, establish optimality.
    \item Changing prices or utilities can change the preferred fruit.
  \end{itemize}

  \vspace{0.5em}
  This comparison will also explain the \hilite{value of an additional dollar}.
  \lecturesource{Worked allocation: utility per dollar and parameter dependence.}
\end{frame}

% Notebook: Minimum-Cost Network Flow Problems as Linear Programs.
\begin{frame}{Minimum-Cost Network Flow}
  Several routings can deliver the same amount of flow at different costs.
  We now seek the \hilite{least expensive routing of a required delivery}.

  \vspace{0.7em}
  Use a finite directed graph $\mathcal G=(\mathcal V,\mathcal E)$,
  distinct source $s$ and sink $t$, and no self-loops. Fix vertex and edge order.
  \begin{itemize}\setlength{\itemsep}{0.65em}
    \item $f_j$: flow on edge $j$; $c_j\geq0$: its finite capacity.
    \item $w_j$: cost per unit of flow on edge $j$.
    \item $F\geq0$: the specified net flow from source to sink.
  \end{itemize}

  \vspace{0.5em}
  Flows, capacities, and $F$ are measured in the same units.
  \lecturesource{Minimum-Cost Network Flow Problems as Linear Programs: assumptions and quantities.}
\end{frame}

\begin{frame}{An Incidence Matrix Encodes Conservation}
  The matrix $\mathbf A\in\mathbb R^{|\mathcal V|\times|\mathcal E|}$
  has one row per vertex $i$ and one column per edge $j$.
  Its entries and required net inflows $b_i$ are:

  \vspace{0.5em}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.55\textwidth}
      \[
        A_{i,j}=\begin{cases}
          1,&j\text{ enters }i,\\
          -1,&j\text{ leaves }i,\\
          0,&\text{otherwise}.
        \end{cases}
      \]
    \end{column}
    \begin{column}{0.41\textwidth}
      \[
        b_i=\begin{cases}
          -F,&\text{source},\\
          F,&\text{sink},\\
          0,&\text{otherwise}.
        \end{cases}
      \]
    \end{column}
  \end{columns}

  \vspace{0.7em}
  Collect flows in $\mathbf f\in\mathbb R^{|\mathcal E|}$ and net inflows
  in $\mathbf b\in\mathbb R^{|\mathcal V|}$. Each row enforces:
  \[
    \underbrace{(\mathbf A\mathbf f)_i}_{\text{inflow minus outflow}}=b_i.
  \]
  \lecturesource{Node-edge incidence convention and required net-inflow vector.}
\end{frame}

\begin{frame}{A Linear Program for the Required Delivery}
  Encode the required flow $F$ in $\mathbf b$ and minimize total edge cost:
  \[
    \begin{aligned}
      \underset{\mathbf f}{\text{minimize}}\quad
        & \sum_{j\in\mathcal E} w_j f_j\\
      \text{subject to}\quad & \mathbf A\mathbf f=\mathbf b,\\
        & 0\leq f_j\leq c_j,\qquad j\in\mathcal E.
    \end{aligned}
  \]

  \vspace{0.6em}
  \begin{itemize}\setlength{\itemsep}{0.7em}
    \item Equalities enforce delivery and conservation.
    \item Bounds enforce nonnegative flows and edge capacities.
    \item Costs distinguish feasible routings with the same delivery.
  \end{itemize}
  This formulation applies to general directed networks.
  \lecturesource{Minimum-cost flow for a specified delivery.}
\end{frame}

\begin{frame}{Worked Network: Two Routes}
  Deliver $F=3$ flow units. Costs are in dollars per flow unit.

  \vspace{0.4em}

  \end{center}

  Through $v$: cost $1+1=2$ dollars per unit, capacity 2.
  Direct: cost 5 dollars per unit.

  \vspace{0.6em}
  Some of the 3-unit delivery must travel directly.
  \lecturesource{Worked Network: Choosing Between Two Routes.
    Original lecture diagram: ../figs/Fig-MinCostFlow-ThreeNode.svg;
    provenance in ../figs/README.md.}
\end{frame}

\begin{frame}{Write the Three Vertex Balances}
  Order rows as $(s,v,t)$ and columns as edges $(1,2,3)$.
  For $F=3$, conservation gives:
  \[
    \underbrace{\begin{bmatrix}-1&0&-1\\1&-1&0\\0&1&1\end{bmatrix}}_{\mathbf A}
    \underbrace{\begin{bmatrix}f_1\\f_2\\f_3\end{bmatrix}}_{\mathbf f}
    =\underbrace{\begin{bmatrix}-3\\0\\3\end{bmatrix}}_{\mathbf b}.
  \]

  \vspace{0.4em}
  \begin{itemize}\setlength{\itemsep}{0.6em}
    \item Source: $f_1+f_3=3$ units leave.
    \item Intermediate vertex: $f_1=f_2$.
    \item Sink: $f_2+f_3=3$ units enter.
  \end{itemize}
  Each edge column contains a $-1$ at its start and a $+1$ at its end.
  \lecturesource{Worked network: ordered incidence matrix and vertex balances.}
\end{frame}

\begin{frame}{Use the Cheaper Route Up to Its Capacity}
  Let $q$ be the flow through $v$. Conservation gives
  $f_1=f_2=q$ and $f_3=3-q$; capacities require $0\leq q\leq2$.

  \vspace{0.5em}
  The total cost, in dollars, is:
  \[
    C(q)=\underbrace{q+q}_{\text{route through }v}
         +\underbrace{5(3-q)}_{\text{direct route}}
         =15-3q.
  \]
  Cost decreases with $q$, so use $q^\star=2$:
  \[
    (f_1^\star,f_2^\star,f_3^\star)=(2,2,1),
    \qquad C^\star=\hilite{9}.
  \]
  Sending all 3 units directly is feasible, but costs 15 dollars.
  Conservation and capacity checks alone do not select the cheapest routing.
  \lecturesource{Worked network: one-variable cost reduction and optimal flow.}
\end{frame}

\begin{frame}{Minimum-Cost Maximum Flow Takes Two Steps}
  Let $F_{\max}$ be the largest feasible net source-to-sink flow.
  To find a minimum-cost maximum flow:

  \vspace{0.5em}
  \begin{enumerate}\setlength{\itemsep}{0.6em}
    \item Solve the maximum-flow problem to find $F_{\max}$.
    \item Set $F=F_{\max}$ and minimize cost among routings delivering it.
  \end{enumerate}

  \vspace{0.4em}
  In our network, source capacities give a bound reached by a feasible flow:
  \[
    F_{\max}=2+3=5,\qquad(f_1,f_2,f_3)=(2,2,3).
  \]
  Delivering 5 units costs 19 dollars; the earlier 9-dollar result delivers
  3 units. A requirement above 5 units is infeasible.

  \slidenote{Next lab:}{\href{\lablink}{L5d uses costs and capacities in a worker-task assignment model.}}
  \lecturesource{Worked network: minimum-cost maximum flow and the L5d connection.}
\end{frame}

% Notebook: Dual Linear Programming Problems; consumer budget first.
\begin{frame}{Dual Linear Programming Problems}
  How can we show that no feasible purchase is better?
  Assign a value $y\geq0$ to each dollar of budget, in utility units per dollar.

  Recall product prices $p_i$, utilities $u_i$, and budget $I$.
  The \hilite{consumer dual} finds the smallest utility bound:
  \[
    \begin{aligned}
      \underset{y}{\text{minimize}}\quad & Iy\\
      \text{subject to}\quad & p_i y\geq u_i,\qquad i=1,\ldots,n,\\
        & y\geq0.
    \end{aligned}
  \]
  Each constraint makes the assigned value of a product's cost cover its utility.
  A feasible purchase spends at most $I$, so its utility is at most $Iy$.
  \lecturesource{What Is the Consumer's Budget Worth? Shortened approved consumer dual.}
\end{frame}

\begin{frame}{What Is the Consumer's Budget Worth?}
  For our apples-and-oranges case, $2y\geq0.55$ and $4y\geq0.45$.
  The smallest feasible value and its budget bound are:
  \[
    y^\star=\max\!\left\{\frac{0.55}{2},\frac{0.45}{4}\right\}=0.275,
    \qquad Iy^\star=27.5.
  \]
  Buying 50 apples attains 27.5 utility units. Matching the bound
  \hilite{proves optimality}.

  \vspace{0.8em}
  The budget's \hilite{shadow price} is $y^\star$:
  an extra dollar buys 0.5 units of apples and adds 0.275 utility units.

  \slidenote{Interpretation:}{fixed prices and utilities, divisible quantities, and sufficient stock. Shadow prices can change as resource availability changes.}
  \lecturesource{Consumer dual: matching bound and shadow-price interpretation for the stated parameters.}
\end{frame}

% Notebook: The General Primal-Dual Pair.
\begin{frame}{The General Primal-Dual Pair}
  Return to $n$ activities and $m$ resources, with
  $\mathbf A\in\mathbb R^{m\times n}$.
  Let $\mathbf y\in\mathbb R^m$ assign a value to each resource.
  The paired models are:

  \vspace{0.4em}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.46\textwidth}
      \textbf{\ink{Primal: choose activities}}
      \[
        \begin{aligned}
          \max_{\mathbf x}\quad & \mathbf c^\top\mathbf x\\
          \text{s.t.}\quad & \mathbf A\mathbf x\leq\mathbf b,\\
            & \mathbf x\geq\mathbf0.
        \end{aligned}
      \]
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{\ink{Dual: value resources}}
      \[
        \begin{aligned}
          \min_{\mathbf y}\quad & \mathbf b^\top\mathbf y\\
          \text{s.t.}\quad & \mathbf A^\top\mathbf y\geq\mathbf c,\\
            & \mathbf y\geq\mathbf0.
        \end{aligned}
      \]
    \end{column}
  \end{columns}

  \vspace{0.8em}
  There is one dual variable per primal resource constraint and one dual
  constraint per primal activity.
  \lecturesource{The General Primal-Dual Pair.}
\end{frame}

\begin{frame}{Resource Values Must Cover Each Activity}
  Recall that $A_{j,i}$ is the amount of resource $j$ used per unit of activity $i$,
  and $c_i$ is the objective contribution per unit of that activity.
  Value resource $j$ at $y_j\geq0$ objective units per resource unit.

  \vspace{0.5em}
  For each activity, the dual requires:
  \[
    \underbrace{\sum_{j=1}^m A_{j,i}y_j}_{\text{assigned resource value}}
    \geq\underbrace{c_i}_{\text{contribution}},
    \qquad i=1,\ldots,n.
  \]
  The transpose combines entries down each activity column of $\mathbf A$.

  \vspace{0.7em}
  The objective $\mathbf b^\top\mathbf y$ totals the assigned value of all
  available resources. The dual seeks the smallest such bound.
  \lecturesource{Interpretation of the scalar dual constraints and the transpose.}
\end{frame}

\begin{frame}{Weak Duality Bounds Every Feasible Decision}
  Let $\mathbf x$ be primal-feasible and let $\mathbf y$ be dual-feasible.
  Their constraints and nonnegative components give:
  \[
    \mathbf c^\top\mathbf x
    \leq(\mathbf A^\top\mathbf y)^\top\mathbf x
    =\mathbf y^\top\mathbf A\mathbf x
    \leq\mathbf y^\top\mathbf b.
  \]

  \vspace{0.6em}
  \begin{itemize}\setlength{\itemsep}{0.8em}
    \item First inequality: $\mathbf A^\top\mathbf y\geq\mathbf c$
      and $\mathbf x\geq\mathbf0$.
    \item Last inequality: $\mathbf A\mathbf x\leq\mathbf b$
      and $\mathbf y\geq\mathbf0$.
  \end{itemize}

  \vspace{0.7em}
  This is \hilite{weak duality}: every feasible dual objective value is an upper
  bound on every feasible primal objective value.
  \lecturesource{Weak-duality derivation.
    Supporting reference: \url{https://courses.csail.mit.edu/6.854/16/Notes/n12-duality.html}.}
\end{frame}

\begin{frame}{A Zero Duality Gap Certifies Optimality}
  For feasible candidates, define the \hilite{duality gap} as:
  \[
    \mathbf b^\top\mathbf y-\mathbf c^\top\mathbf x\geq0.
  \]
  A zero gap proves both candidates are optimal: the primal reaches the
  upper bound supplied by the dual.

  \vspace{0.7em}
  \hilite{Strong duality}: if the primal is feasible with a finite optimal
  value, the dual also has an optimal solution. Optimal solutions satisfy:
  \[
    \mathbf c^\top\mathbf x^\star=\mathbf b^\top\mathbf y^\star.
  \]
  The objective values agree; the decision vectors describe different quantities.
  \lecturesource{Duality gap and strong duality.
    Supporting reference: \url{https://courses.csail.mit.edu/6.854/16/Notes/n12-duality.html}.}
\end{frame}

% Notebook: Constraint Directions and Variable Signs.
\begin{frame}{Constraint Directions and Variable Signs}
  For the \hilite{maximization convention} just used, the bound determines
  the following correspondences:

  \vspace{0.6em}
  \renewcommand{\arraystretch}{1.45}
  \begin{tabularx}{\textwidth}{@{}YY@{}}
    \toprule
    Primal condition & Corresponding dual condition\\
    \midrule
    $\sum_i A_{j,i}x_i\leq b_j$ & $y_j\geq0$\\
    $\sum_i A_{j,i}x_i=b_j$ & $y_j$ free to have either sign\\
    $x_i\geq0$ & $(\mathbf A^\top\mathbf y)_i\geq c_i$\\
    $x_i$ free to have either sign & $(\mathbf A^\top\mathbf y)_i=c_i$\\
    \bottomrule
  \end{tabularx}

  \vspace{0.4em}
  An equality can be multiplied by a value of either sign. A free primal variable
  requires matching coefficients for both positive and negative values.

  \vspace{0.5em}
  Network-flow equality balances therefore have unrestricted dual multipliers.
  \lecturesource{Constraint Directions and Variable Signs.
    Supporting reference: \url{https://courses.csail.mit.edu/6.854/16/Notes/n12-duality.html}.}
\end{frame}

% Notebook: How a Solver Fits the Modeling Workflow.
\begin{frame}{How a Solver Fits the Modeling Workflow}
  Interpret a candidate only after checking what the solver established:

  \vspace{0.7em}
  \begin{enumerate}\setlength{\itemsep}{0.7em}
    \item \hilite{Read the status.} An optimal status reports optimality within
      numerical tolerances. A time or iteration limit does not; first confirm
      a feasible solution is available.
    \item \hilite{Verify feasibility.} Check the original constraints and bounds,
      including every node balance for a flow model.
    \item \hilite{Recompute the objective.} Use the returned quantities and
      original coefficients; compare with the reported value.
    \item \hilite{Check an optimality bound.} Verify dual feasibility before
      comparing primal and dual objective values.
  \end{enumerate}

  \slidenote{Numerical checks:}{use stated tolerances appropriate to the units and scales. See the \href{https://jump.dev/JuMP.jl/stable/manual/solutions/}{JuMP solution guide}.}
  \lecturesource{Solver workflow. External documentation:
    \url{https://jump.dev/JuMP.jl/stable/manual/solutions/}.}
\end{frame}

\begin{frame}{Supporting Algorithm: Revised Simplex}
  How does a solver find a candidate?
  The \href{\simplexlink}{revised-simplex notebook} develops three ideas:

  \vspace{0.8em}
  \begin{itemize}\setlength{\itemsep}{0.8em}
    \item A \hilite{feasible basis} selects constraint columns that determine

    \item \hilite{Reduced costs} account for objective changes when a nonbasic

    \item The \hilite{ratio test} limits the step so that the updated solution
      remains feasible.
  \end{itemize}
