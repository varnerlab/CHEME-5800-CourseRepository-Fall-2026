% 5->9, 6->10, 9->13, 10->13. Edge (1,3) keeps capacity one but carries no
% flow; it lies inside S and does not cross the cut.

  \draw[partition,draw=vnquiet!55,fill=black!2]
    (1.75,-1.9) rectangle (10.15,2.25);
  \draw[partition,draw=vnslidelink!55,fill=white]
    (-0.55,-0.48) rectangle (2.85,0.48);

  \coordinate (t5) at (4.8,1.35);
  \coordinate (t6) at (4.8,0.45);
